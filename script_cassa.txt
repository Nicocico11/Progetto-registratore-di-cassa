===== processa_ia.py =====
import json, re, csv, datetime, os, sys

file_path = os.path.expanduser('~/.termux/tasker/output_ia.txt')
testo_originale = sys.argv[1] if len(sys.argv) > 1 else "Non specificato"

if not os.path.exists(file_path):
    print("❌ File output non trovato.")
    sys.exit(1)

with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
    text = f.read()

text_clean = text.replace('```json', '').replace('```', '')
matches = re.findall(r'\{.*?\}', text_clean, re.DOTALL)

data = {}
importo_numerico = None

# Tentativo 1: Parsing JSON standard dall'ultimo blocco valido
if matches:
    raw = matches[-1]
    clean = re.sub(r'\x1b\[[0-9;]*[a-zA-Z]', '', raw)
    clean = clean.replace('\n', ' ').replace('\r', ' ').replace('\t', ' ')
    clean = "".join(c for c in clean if ord(c) >= 32)
    try:
        data = json.loads(clean)
    except:
        pass

# Tentativo 2 (Fallback d'emergenza): Estrazione via Regex se il JSON fallisce o è troncato
if not data or 'importo' not in data:
    importo_match = re.search(r'["\']?importo["\']?\s*:\s*["\']?([0-9]+\.?[0-9]*)["\']?', text_clean)
    if importo_match:
        imp_str = importo_match.group(1)
    else:
        num_match = re.search(r'\d+', testo_originale)
        imp_str = num_match.group(0) if num_match else "0"

    data = {
        "categoria": "Gasolio" if "gasolio" in testo_originale.lower() else "Benzina",
        "metodo_pagamento": "Carta" if "carta" in testo_originale.lower() or "pos" in testo_originale.lower() else "Contanti",
        "importo": imp_str,
        "note": testo_originale
    }

# --- INTEGRAZIONE LISTINO PREZZI.JSON ---
prezzi_path = os.path.expanduser('~/prezzi.json')
if os.path.exists(prezzi_path):
    try:
        with open(prezzi_path, 'r', encoding='utf-8') as pf:
            listino = json.load(pf)
            testo_basso = testo_originale.lower()
            
            for prod_key, prezzo_unitario in listino.items():
                parole_chiave = prod_key.replace('_', ' ').split()
                if any(parola in testo_basso for parola in parole_chiave):
                    m_num = re.search(r'(\d+(?:[.,]\d+)?)', testo_basso)
                    if m_num:
                        qta_str = m_num.group(1).replace(',', '.')
                        qta = float(qta_str)
                        calcolato = qta * float(prezzo_unitario)
                        data['importo'] = f"{calcolato:.2f}"
                        data['categoria'] = prod_key.replace('_', ' ').title()
                        break
    except Exception as e:
        print(f"⚠️ Errore lettura prezzi.json: {e}")
# ----------------------------------------

try:
    importo_str = str(data.get('importo', '0'))
    importo_numerico = re.sub(r'[^0-9.]', '', importo_str)

    if not importo_numerico or float(importo_numerico) == 0 or '123.45' in importo_numerico:
        print("❌ Transazione scartata: Nessun importo valido rilevato.")
        sys.exit(1)

    now = datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')
    csv_file = os.path.expanduser('~/transazioni_turno.csv')

    file_exists = os.path.isfile(csv_file)
    with open(csv_file, 'a', newline='', encoding='utf-8') as f:
        writer = csv.writer(f)
        if not file_exists:
            writer.writerow(['data_ora', 'dettagli_json', 'importo'])
        writer.writerow([now, json.dumps(data), float(importo_numerico)])

    print('✅ Vendita salvata:', data)
except Exception as e:
    print('❌ Errore fatale nel salvataggio:', e)
